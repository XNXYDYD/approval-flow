<script lang="ts">
  import { LABEL_MAP } from '$lib/types';
  import type { OvertimeApplication } from '$lib/types';
  import { formatDuration } from '$lib/utils/duration';
  import { createEventDispatcher } from 'svelte';
  import { Button } from '$lib/components/ui/button';
  import Pencil from 'lucide-svelte/icons/pencil';
  import CheckCircle2 from 'lucide-svelte/icons/check-circle-2';
  import X from 'lucide-svelte/icons/x';

  export let application: Partial<OvertimeApplication>;
  const dispatch = createEventDispatcher();

  // 格式化时间用于展示
  function formatTime(iso: string): string {
    if (!iso) return '-';
    const d = new Date(iso);
    return d.toLocaleString('zh-CN', {
      year: 'numeric', month: '2-digit', day: '2-digit',
      hour: '2-digit', minute: '2-digit'
    });
  }
</script>

<div class="p-6 bg-white rounded-lg shadow space-y-6">
  <h2 class="text-xl font-bold border-b pb-3">加班申请预览</h2>

  <section class="space-y-2">
    <h3 class="font-medium text-gray-800 mb-2">申请人信息</h3>
    <div class="grid grid-cols-2 gap-2 text-sm">
      <p><span class="text-gray-500">姓名：</span>{application.applicant?.name ?? '-'}</p>
      <p><span class="text-gray-500">工号：</span>{application.applicant?.employeeId ?? '-'}</p>
      <p><span class="text-gray-500">部门：</span>{application.applicant?.department?.name ?? '-'}</p>
    </div>
  </section>

  <section class="space-y-2">
    <h3 class="font-medium text-gray-800 mb-2">加班内容</h3>
    <div class="grid grid-cols-2 gap-2 text-sm">
      <p><span class="text-gray-500">加班类型：</span>{LABEL_MAP[application.overtimeType ?? ''] ?? '-'}</p>
      <p><span class="text-gray-500">补偿方式：</span>{LABEL_MAP[application.compensation ?? ''] ?? '-'}</p>
      <p><span class="text-gray-500">开始时间：</span>{formatTime(application.startTime ?? '')}</p>
      <p><span class="text-gray-500">结束时间：</span>{formatTime(application.endTime ?? '')}</p>
      <p class="col-span-2"><span class="text-gray-500">加班时长：</span>{formatDuration(application.duration ?? 0)}</p>
    </div>
    <div class="mt-3 p-3 bg-gray-50 rounded-md text-sm">
      <span class="text-gray-500">加班事由：</span>{application.reason ?? '-'}
    </div>
  </section>

  <div class="flex gap-3 pt-4 border-t">
    <Button variant="outline" on:click={() => dispatch('edit')}>
      <Pencil class="mr-2 h-4 w-4" />
      返回修改
    </Button>
    <Button on:click={() => dispatch('submit')} class="bg-green-600 hover:bg-green-700">
      <CheckCircle2 class="mr-2 h-4 w-4" />
      确认提交
    </Button>
    <Button variant="outline" on:click={() => dispatch('cancel')}>
      <X class="mr-2 h-4 w-4" />
      取消
    </Button>
  </div>
</div>
